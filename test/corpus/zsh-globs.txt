env_loader_loop() {
    local -a files

    files=("$SRC_CFG_DIR"/*.sh(N))

    if ((${#files[@]} == 0)); then
        msg --warn "no *.sh files found in '$SRC_CFG_DIR'"
        return 0
    fi
}
